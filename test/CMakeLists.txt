cmake_minimum_required(VERSION 3.20)
project(Dx12UtilsTests LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    if(MSVC)
        set(PROFILE_COMPILE_OPTIONS /Od /Zi /fsanitize=address)
        set(PROFILE_LINK_OPTIONS)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        set(PROFILE_COMPILE_OPTIONS -O0 -g -fsanitize=address)
        set(PROFILE_LINK_OPTIONS -fsanitize=address)
    else()
        message(FATAL_ERROR "Debug builds require an MSVC, GNU, or Clang C++ compiler.")
    endif()
elseif(CMAKE_BUILD_TYPE STREQUAL "Release")
    if(MSVC)
        set(PROFILE_COMPILE_OPTIONS /O2)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        set(PROFILE_COMPILE_OPTIONS -O2)
    else()
        message(FATAL_ERROR "Release builds require an MSVC, GNU, or Clang C++ compiler.")
    endif()
    set(PROFILE_LINK_OPTIONS)
else()
    message(FATAL_ERROR
        "CMAKE_BUILD_TYPE must be Debug or Release. Configure using one of the VS Code launch profiles.")
endif()

add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_BINARY_DIR}/Dx12Utils")

add_executable(Dx12UtilsTests test.cpp)
target_link_libraries(Dx12UtilsTests PRIVATE Dx12Utils)

foreach(TARGET Dx12Utils Dx12UtilsTests)
    target_compile_options(${TARGET} PRIVATE ${PROFILE_COMPILE_OPTIONS})
    target_link_options(${TARGET} PRIVATE ${PROFILE_LINK_OPTIONS})
endforeach()

if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_compile_definitions(Dx12Utils PRIVATE NDEBUG)
    target_compile_definitions(Dx12UtilsTests PRIVATE NDEBUG)
endif()

include(CTest)
if(BUILD_TESTING)
    add_test(NAME Dx12UtilsTests COMMAND Dx12UtilsTests)
endif()
